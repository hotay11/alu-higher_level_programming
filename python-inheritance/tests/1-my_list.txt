>>> MyList = __import__('1-my_list').MyList

# Basic usage
>>> my_list = MyList()
>>> my_list.append(1)
>>> my_list.append(4)
>>> my_list.append(2)
>>> my_list.print_sorted()
[1, 2, 4]

# Original list unchanged
>>> print(my_list)
[1, 4, 2]

# Empty list
>>> empty = MyList()
>>> empty.print_sorted()
[]

# Single element
>>> one = MyList()
>>> one.append(5)
>>> one.print_sorted()
[5]

# Already sorted
>>> sorted_list = MyList()
>>> sorted_list.append(1)
>>> sorted_list.append(2)
>>> sorted_list.append(3)
>>> sorted_list.print_sorted()
[1, 2, 3]

# Negative numbers
>>> neg = MyList()
>>> neg.append(-3)
>>> neg.append(0)
>>> neg.append(-1)
>>> neg.print_sorted()
[-3, -1, 0]
